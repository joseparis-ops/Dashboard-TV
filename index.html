  <!-- Lógica JavaScript -->

  <script>

    // CREDENCIALES DE TU PROYECTO SUPABASE

    const SUPABASE_URL = 'https://sotvmiyeinoldidpvyhd.supabase.co';[cite: 1]

    const SUPABASE_KEY = 'sb_publishable_QK06zg7-6l6wqitqfGPEfA_zimMK...'; // Pega tu Publishable Key completa[cite: 3]

    const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);


    let rawData = [];

    let currentUserRole = 'viewer';

    let chartProgramasInstance = null;

    let chartCanalesInstance = null;


    // 1. Validar Sesión y Cargar Rol

    async function initDashboard() {

      const { data: { session } } = await supabaseClient.auth.getSession();

      if (!session) {

        window.location.href = 'login.html';

        return;

      }


      // Obtener el rol del usuario

      const { data: profile } = await supabaseClient

        .from('profiles')

        .select('role')

        .eq('id', session.user.id)

        .single();


      if (profile) {

        currentUserRole = profile.role;

        document.getElementById('role-badge').textContent = `Rol: ${currentUserRole.toUpperCase()}`;

        if (currentUserRole === 'admin') {

          document.querySelectorAll('.admin-only').forEach(el => el.classList.remove('hidden'));

        }

      }


      await fetchData();

    }


    // 2. Traer Datos de Supabase

    async function fetchData() {

      const { data, error } = await supabaseClient

        .from('tv_measurements')

        .select('*')

        .order('created_at', { ascending: false });


      if (error) {

        console.error('Error al obtener mediciones:', error);

        return;

      }


      rawData = data || [];

      populateFilterOptions();

      applyFilters();

    }


    // 3. Poblar Filtros

    function populateFilterOptions() {

      const canales = [...new Set(rawData.map(d => d.canal))];

      const programas = [...new Set(rawData.map(d => d.programa))];

      const meses = [...new Set(rawData.map(d => d.mes))];


      fillSelect('filter-canal', canales, 'Todos los canales');

      fillSelect('filter-programa', programas, 'Todos los programas');

      fillSelect('filter-mes', meses, 'Todos los meses');

    }


    function fillSelect(elementId, items, defaultText) {

      const select = document.getElementById(elementId);

      select.innerHTML = `<option value="ALL">${defaultText}</option>`;

      items.forEach(item => {

        select.innerHTML += `<option value="${item}">${item}</option>`;

      });

    }


    // 4. Aplicar Filtros

    function applyFilters() {

      const canalVal = document.getElementById('filter-canal').value;

      const programaVal = document.getElementById('filter-programa').value;

      const diaVal = document.getElementById('filter-dia').value;

      const mesVal = document.getElementById('filter-mes').value;


      const filtered = rawData.filter(d => {

        return (canalVal === 'ALL' || d.canal === canalVal) &&

               (programaVal === 'ALL' || d.programa === programaVal) &&

               (diaVal === 'ALL' || d.dia === diaVal) &&

               (mesVal === 'ALL' || d.mes === mesVal);

      });


      renderKPIs(filtered);

      renderCharts(filtered);

      renderTable(filtered);

    }


    // 5. Renderizar KPIs

    function renderKPIs(data) {

      if (!data.length) {

        document.getElementById('kpi-rating').textContent = '0.0 pts';

        document.getElementById('kpi-share').textContent = '0.0%';

        document.getElementById('kpi-reach').textContent = '0 m';

        document.getElementById('kpi-fidelidad').textContent = '0 min';

        return;

      }


      const avgRating = (data.reduce((acc, curr) => acc + Number(curr.rating), 0) / data.length).toFixed(1);

      const avgShare = (data.reduce((acc, curr) => acc + Number(curr.share), 0) / data.length).toFixed(1);


      document.getElementById('kpi-rating').innerHTML = `${avgRating} <span class="text-sm font-normal text-slate-400">pts</span>`;

      document.getElementById('kpi-share').innerHTML = `${avgShare}<span class="text-sm font-normal text-slate-400">%</span>`;

      document.getElementById('kpi-reach').innerHTML = `${(data.length * 150)} <span class="text-sm font-normal text-slate-400">m</span>`;

      document.getElementById('kpi-fidelidad').innerHTML = `35 <span class="text-sm font-normal text-slate-400">min</span>`;

    }


    // 6. Renderizar Gráficos Chart.js

    function renderCharts(data) {

      const programasMap = {};

      const canalesMap = {};


      data.forEach(d => {

        programasMap[d.programa] = (programasMap[d.programa] || 0) + Number(d.rating);

        canalesMap[d.canal] = (canalesMap[d.canal] || 0) + Number(d.share);

      });


      // Gráfico de Programas

      const ctxProg = document.getElementById('chartProgramas').getContext('2d');

      if (chartProgramasInstance) chartProgramasInstance.destroy();

      chartProgramasInstance = new Chart(ctxProg, {

        type: 'bar',

        data: {

          labels: Object.keys(programasMap),

          datasets: [{

            label: 'Rating Promedio',

            data: Object.values(programasMap),

            backgroundColor: '#0284c7',

            borderRadius: 8

          }]

        },

        options: { responsive: true, maintainAspectRatio: false }

      });


      // Gráfico de Canales

      const ctxCanal = document.getElementById('chartCanales').getContext('2d');

      if (chartCanalesInstance) chartCanalesInstance.destroy();

      chartCanalesInstance = new Chart(ctxCanal, {

        type: 'doughnut',

        data: {

          labels: Object.keys(canalesMap),

          datasets: [{

            data: Object.values(canalesMap),

            backgroundColor: ['#0284c7', '#10b981', '#f59e0b', '#ec4899', '#8b5cf6']

          }]

        },

        options: { responsive: true, maintainAspectRatio: false }

      });

    }


    // 7. Renderizar Tabla

    function renderTable(data) {

      const tbody = document.getElementById('table-body');

      tbody.innerHTML = '';


      if (!data.length) {

        tbody.innerHTML = '<tr><td colspan="7" class="text-center py-6 text-slate-500">No hay datos registrados</td></tr>';

        return;

      }


      data.forEach(row => {

        tbody.innerHTML += `

          <tr class="hover:bg-slate-700/30 transition">

            <td class="py-3 px-4 font-semibold text-white">${row.canal}</td>

            <td class="py-3 px-4">${row.programa}</td>

            <td class="py-3 px-4 text-slate-400">${row.dia}</td>

            <td class="py-3 px-4 text-slate-400">${row.mes}</td>

            <td class="py-3 px-4 text-right font-bold text-blue-400">${row.rating} pts</td>

            <td class="py-3 px-4 text-right font-bold text-emerald-400">${row.share}%</td>

            ${currentUserRole === 'admin' ? `

              <td class="py-3 px-4 text-right">

                <button onclick="deleteRecord(${row.id})" class="text-red-400 hover:text-red-300 p-1" title="Eliminar">

                  <i class="fa-solid fa-trash text-xs"></i>

                </button>

              </td>

            ` : ''}

          </tr>

        `;

      });

    }


    // 8. Funciones de Edición/Carga (Solo ADMIN)

    function openModal() { document.getElementById('modal-form').classList.remove('hidden'); }

    function closeModal() { document.getElementById('modal-form').classList.add('hidden'); }


    async function saveMeasurement(e) {

      e.preventDefault();

      const newRecord = {

        canal: document.getElementById('input-canal').value,

        programa: document.getElementById('input-programa').value,

        dia: document.getElementById('input-dia').value,

        mes: document.getElementById('input-mes').value,

        rating: parseFloat(document.getElementById('input-rating').value),

        share: parseFloat(document.getElementById('input-share').value)

      };


      const { error } = await supabaseClient.from('tv_measurements').insert([newRecord]);

      if (error) {

        alert('Error al guardar registro: ' + error.message);

      } else {

        closeModal();

        document.getElementById('add-form').reset();

        await fetchData();

      }

    }


    async function deleteRecord(id) {

      if (!confirm('¿Estás seguro de eliminar este registro?')) return;

      const { error } = await supabaseClient.from('tv_measurements').delete().eq('id', id);

      if (!error) await fetchData();

    }


    function resetFilters() {

      document.getElementById('filter-canal').value = 'ALL';

      document.getElementById('filter-programa').value = 'ALL';

      document.getElementById('filter-dia').value = 'ALL';

      document.getElementById('filter-mes').value = 'ALL';

      applyFilters();

    }


    async function logout() {

      await supabaseClient.auth.signOut();

      window.location.href = 'login.html';

    }


    initDashboard();

  </script>
